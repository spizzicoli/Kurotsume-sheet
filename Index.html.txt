<script>
function creaPallini(id, n) {
    let c = document.getElementById(id);
    c.innerHTML = "";
    for (let i = 0; i < n; i++) {
        let p = document.createElement("div");
        p.className = localStorage.getItem(id+"_"+i) === "1" ? "attivo" : "";
        p.onclick = () => {
            p.classList.toggle("attivo");
            localStorage.setItem(id+"_"+i, p.classList.contains("attivo") ? "1" : "0");
        };
        c.appendChild(p);
    }
}
</script>

// ===============================
// CALCOLI AUTOMATICI
// ===============================
function calc() {
    let f = parseInt(for.value);
    let d = parseInt(des.value);
    let c = parseInt(cos.value);
    let ca = parseInt(car.value);
    let bcVal = parseInt(bc.value);

    let modF = Math.floor((f-10)/2);
    let modD = Math.floor((d-10)/2);
    let modC = Math.floor((c-10)/2);
    let modCa = Math.floor((ca-10)/2);

    modFor.innerText = modF;
    modDes.innerText = modD;
    modCos.innerText = modC;
    modCar.innerText = modCa;

    // CA = armatura completa +1 (18) + Des (max 2)
    ca.innerText = 18 + modD;

    init.innerText = modD;

    cdSpell.innerText = 8 + bcVal + modCa;
    atkSpell.innerText = bcVal + modCa;

    atkKatana.innerText = modF + bcVal + 3; // arma +3
    dmgKatana.innerText = "1d8+" + (modF + 3 + 2); // duellante +2
}

document.querySelectorAll("input").forEach(i => {
    i.oninput = () => {
        localStorage.setItem(i.id, i.value);
        calc();
    };
});

// Carica dati salvati
window.onload = () => {
    document.querySelectorAll("input").forEach(i => {
        if(localStorage.getItem(i.id)) i.value = localStorage.getItem(i.id);
    });
    calc();
};
</script>
